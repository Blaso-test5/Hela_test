const SHA256_RE = /^[0-9a-f]{64}$/;

export function sha256Buffer(buffer) {
  return createHash("sha256").update(buffer).digest("hex");
}

export function parseLoadedOverrides(logText) {
  const loaded = new Map();
  const re = /SHADER READY\s+stage=(?:VS|PS|GS|HS|DS|CS)\s+hash=[0-9a-fA-F]{16}\s+source=SM5-override\s+path=(.+?)\s+bytes=\d+\s+sha256=([0-9a-fA-F]{64})/g;
  let match;
  while ((match = re.exec(logText))) {
    const fileName = match[1].replaceAll("\\", "/").split("/").pop();
    loaded.set(fileName, match[2].toLowerCase());
  }
  return loaded;
}

export function bmpNonBlackFraction(buffer) {
  if (buffer.length < 54 || buffer.toString("ascii", 0, 2) !== "BM") return null;
  const offset = buffer.readUInt32LE(10);
  const width = buffer.readInt32LE(18);
  const height = Math.abs(buffer.readInt32LE(22));
  const bpp = buffer.readUInt16LE(28);
  const compression = buffer.readUInt32LE(30);
  if (width <= 0 || height <= 0 || (bpp !== 24 && bpp !== 32) || compression !== 0) return null;
  const bytesPerPixel = bpp / 8;
  const rowStride = Math.ceil((width * bytesPerPixel) / 4) * 4;
  if (buffer.length < offset + rowStride * height) return null;
  let nonBlack = 0;
  for (let y = 0; y < height; y++) {
    const row = offset + y * rowStride;
    for (let x = 0; x < width; x++) {
      const pixel = row + x * bytesPerPixel;
      if ((buffer[pixel] | buffer[pixel + 1] | buffer[pixel + 2]) !== 0) nonBlack++;
    }
  }
  return nonBlack / (width * height);
}

async function readJson(absPath) {
  try {
    return JSON.parse((await readFile(absPath, "utf8")).replace(/^\uFEFF/, ""));
  } catch (error) {
    if (error?.code === "ENOENT") return null;
    if (error instanceof SyntaxError) return { __invalidJson: true };
    throw error;
  }
}

async function fileRef(baseDir, relativePath) {
  const buffer = await readFile(path.join(baseDir, relativePath));
  return { path: relativePath, sha256: sha256Buffer(buffer) };
}

export function validateRunEvidence(bundle) {
  const issues = [];
  const add = (code, message) => issues.push({ code, message });
  const isRef = (value) =>
    value && typeof value === "object" &&
    typeof value.path === "string" && value.path.length > 0 &&
    typeof value.sha256 === "string" && SHA256_RE.test(value.sha256);

  if (!bundle || typeof bundle !== "object") return [{ code: "BUNDLE_INVALID", message: "bundle non oggetto" }];
  if (bundle.schemaVersion !== 1) add("BUNDLE_INVALID", "schemaVersion deve essere 1");
  if (typeof bundle.runId !== "string" || !bundle.runId) add("RUN_ID_MISSING", "runId mancante");
  if (!isRef(bundle.package?.manifest) || !isRef(bundle.package?.inventory) || !isRef(bundle.package?.packageReport)) {
    add("BUNDLE_INVALID", "package fileRef non valido");
  }
  if (!isRef(bundle.configuration?.managerRun)) add("BUNDLE_INVALID", "managerRun non valido");
  if (!Array.isArray(bundle.loadedOverrides?.entries)) add("BUNDLE_INVALID", "loadedOverrides.entries non array");
  if (!Array.isArray(bundle.readback)) add("BUNDLE_INVALID", "readback non array");
  if (!bundle.visual || !("menuChecklist" in bundle.visual)) add("BUNDLE_INVALID", "visual.menuChecklist mancante");
  if (!["unverified", "failed", "verified-with-checklist"].includes(bundle.verdict?.menu)) {
    add("BUNDLE_INVALID", "verdict.menu non valido");
  }
  if (bundle.verdict?.menu === "verified-with-checklist") {
    const checklist = bundle.visual?.menuChecklist;
    const allPass = Array.isArray(checklist) && checklist.length > 0 && checklist.every((entry) => entry?.status === "pass");
    if (!Array.isArray(bundle.readback) || bundle.readback.length === 0) add("VERDICT_GUARDRAIL", "serve almeno un readback");
    if (!allPass) add("VERDICT_GUARDRAIL", "serve checklist tutta pass");
    if (!Array.isArray(bundle.loadedOverrides?.entries) || bundle.loadedOverrides.entries.length === 0) {
      add("VERDICT_GUARDRAIL", "serve almeno un override caricato");
    }
  }
  return issues;
}

async function findBmpFiles(runDir) {
  const dir = path.join(runDir, "readbacks");
  try {
    return (await readdir(dir)).filter((name) => /^HelaReadback_Present\d+_(source|backbuffer)\.bmp$/i.test(name)).sort();
  } catch (error) {
    if (error?.code === "ENOENT") return [];
    throw error;
  }
}
