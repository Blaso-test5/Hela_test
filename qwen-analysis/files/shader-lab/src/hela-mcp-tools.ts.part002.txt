      onlyHashes: z.array(z.string()).optional().describe("Subset opzionale di hash FNV-1a64"),
      dxcPath: z.string().optional().describe("Override locale di dxc.exe"),
    },
  }, async (args) => {

    } catch (e) {
      return { isError: true, content: [{ type: "text" as const, text: "build_hela_override_stage: " + String(e) }] };
    }
  });

  server.registerTool("plan_hela_manifest_patch", {
    title: "Pianifica manifest candidato hela-run-manifest/v1",
